function(matter_source_list key prefix suffix output)
  string(JSON count LENGTH "${MATTER_SOURCE_MANIFEST}" ${key})
  math(EXPR last "${count}-1")
  set(result)
  foreach(i RANGE 0 ${last})
    string(JSON name GET "${MATTER_SOURCE_MANIFEST}" ${key} ${i})
    list(APPEND result "${prefix}${name}${suffix}")
  endforeach()
  set(${output} "${result}" PARENT_SCOPE)
endfunction()

function(matter_compile target)
  target_compile_features(${target} PUBLIC c_std_99)
  set_target_properties(${target} PROPERTIES C_EXTENSIONS OFF)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -ffunction-sections -fdata-sections)
  # Upstream mbedTLS supplies its own feature-test macros (entropy_poll.c).
  if(NOT WIN32 AND NOT target STREQUAL "matter_mbedtls")
    target_compile_definitions(${target} PRIVATE _GNU_SOURCE)
  endif()
  if(MATTER_SANITIZE)
    target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined -fno-sanitize-recover=all)
  endif()
endfunction()

if(MATTER_BUILD_HOST)
  if(NOT MATTER_BUILD_RUNTIME)
    message(FATAL_ERROR "Host requires MATTER_BUILD_RUNTIME")
  endif()
  if(WIN32)
    matter_source_list(host_windows "${host_dir}/" .c host_sources)
  else()
    matter_source_list(host_linux "${host_dir}/" .c host_sources)
  endif()
  add_library(matter_host STATIC ${host_sources})
  matter_compile(matter_host)
  target_include_directories(matter_host PUBLIC "${host_dir}" PRIVATE "${MATTER_MBEDTLS_DIR}/include")
  target_compile_definitions(matter_host PRIVATE MBEDTLS_CONFIG_FILE="matter_mbedtls_config.h")
  target_link_libraries(matter_host PUBLIC matter_runtime)
  # UTC build floor for the opt-in per-instance host clock; honors SOURCE_DATE_EPOCH.
  string(TIMESTAMP matter_build_unix "%s" UTC)
  target_compile_definitions(matter_host PUBLIC MATTER_HOST_BUILD_UNIX=${matter_build_unix}ULL)
  if(WIN32)
    target_link_libraries(matter_host PUBLIC ws2_32 iphlpapi bcrypt)
  endif()
  add_executable(matter-node "${host_dir}/matter_node.c")
  matter_compile(matter-node)
  foreach(field VENDOR_NAME PRODUCT_NAME)
    set(value "${MATTER_HOST_${field}}")
    string(LENGTH "${value}" size)
    if(size LESS 1 OR size GREATER 32 OR value MATCHES "[;\n\r]")
      message(FATAL_ERROR "MATTER_HOST_${field} must contain 1..32 UTF-8 bytes without semicolons or line breaks")
    endif()
    string(REPLACE "\\" "\\\\" value "${value}")
    string(REPLACE "\"" "\\\"" value "${value}")
    target_compile_definitions(matter-node PRIVATE "MATTER_HOST_${field}=\"${value}\"")
  endforeach()
  target_link_libraries(matter-node PRIVATE matter_host)
  target_link_options(matter-node PRIVATE -Wl,--gc-sections)
  if(WIN32)
    enable_language(RC)
    set_property(SOURCE "${host_dir}/matter_gui.rc" APPEND PROPERTY OBJECT_DEPENDS
      "${host_dir}/assets/matter-desktop.ico" "${host_dir}/matter_gui.manifest")
    add_executable(matter-desktop WIN32 "${host_dir}/matter_node.c"
      "${host_dir}/matter_gui_windows.c"
      "${host_dir}/matter_gui.rc"
      "${CMAKE_CURRENT_SOURCE_DIR}/third_party/qrcodegen/qrcodegen.c")
    matter_compile(matter-desktop)
    get_target_property(host_identity_definitions matter-node COMPILE_DEFINITIONS)
    target_compile_definitions(matter-desktop PRIVATE ${host_identity_definitions})
    target_compile_definitions(matter-desktop PRIVATE main=matter_node_main _WIN32_WINNT=0x0602)
    target_include_directories(matter-desktop PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/qrcodegen")
    target_link_libraries(matter-desktop PRIVATE matter_host comctl32 shell32 ole32 uuid gdi32 comdlg32)
    target_link_options(matter-desktop PRIVATE -static -Wl,--gc-sections -Wl,--stack,8388608)
    install(TARGETS matter-desktop RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
  endif()
  if(WIN32)
    target_link_options(matter-node PRIVATE -Wl,--stack,8388608)
  endif()
endif()
