set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/cmake/test-cases.json")
file(READ "${PROJECT_SOURCE_DIR}/cmake/test-cases.json" cases)
string(JSON count LENGTH "${cases}")
math(EXPR last "${count}-1")
foreach(i RANGE 0 ${last})
  foreach(key name source mode platform privileged variant)
    string(JSON ${key} GET "${cases}" ${i} ${key})
  endforeach()
  if((platform STREQUAL "windows" AND NOT WIN32) OR (platform STREQUAL "linux" AND WIN32))
    continue()
  endif()
  if(privileged AND NOT MATTER_ENABLE_PRIVILEGED_TESTS)
    continue()
  endif()
  if(privileged AND NOT EXISTS "$ENV{MATTER_IPV6_INJECTOR}")
    message(FATAL_ERROR "MATTER_ENABLE_PRIVILEGED_TESTS requires MATTER_IPV6_INJECTOR")
  endif()
  add_executable(${name} "${PROJECT_SOURCE_DIR}/${source}")
  matter_compile(${name})
  target_compile_options(${name} PRIVATE -UNDEBUG)
  set(test_library matter_host)
  if(name MATCHES "^test_(ack|exchange|rx|retransmission|retransmission_sender)(_alloc)?$")
    set(test_library matter_unit8)
  endif()
  target_link_libraries(${name} PRIVATE ${test_library})
  target_link_options(${name} PRIVATE -Wl,--gc-sections)
  if(WIN32)
    target_link_options(${name} PRIVATE -Wl,--stack,8388608)
  endif()
  if(variant STREQUAL "alloc")
    string(JSON size LENGTH "${cases}" ${i} replace)
    math(EXPR stop "${size}-1")
    set(replacements)
    foreach(j RANGE 0 ${stop})
      string(JSON replacement GET "${cases}" ${i} replace ${j})
      if(WIN32)
        string(REPLACE "@udp@" "matter_udp_windows" replacement "${replacement}")
      else()
        string(REPLACE "@udp@" "matter_udp_linux" replacement "${replacement}")
      endif()
      list(APPEND replacements "${PROJECT_SOURCE_DIR}/${replacement}")
    endforeach()
    add_library(${name}_objects OBJECT ${replacements})
    matter_compile(${name}_objects)
    target_link_libraries(${name}_objects PRIVATE ${test_library})
    target_include_directories(${name}_objects PRIVATE "${MATTER_MBEDTLS_DIR}/include")
    target_compile_definitions(${name}_objects PRIVATE MBEDTLS_CONFIG_FILE="matter_mbedtls_config.h")
    string(JSON malloc ERROR_VARIABLE no_malloc GET "${cases}" ${i} malloc)
    if(NOT no_malloc AND malloc)
      target_compile_definitions(${name}_objects PRIVATE calloc=matter_test_calloc malloc=matter_test_malloc free=matter_test_free)
    else()
      target_compile_definitions(${name}_objects PRIVATE MATTER_TEST_ALLOC_REPLACE)
      target_compile_options(${name}_objects PRIVATE -include "${PROJECT_SOURCE_DIR}/tools/matter/tests/alloc.h")
    endif()
    target_sources(${name} PRIVATE $<TARGET_OBJECTS:${name}_objects>)
    target_compile_definitions(${name} PRIVATE MATTER_TEST_ALLOC)
  endif()
  add_test(NAME ${name} COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tools/matter/ctest_case.py"
    --exe $<TARGET_FILE:${name}> --mode ${mode})
  set_tests_properties(${name} PROPERTIES LABELS "native;${variant}" TIMEOUT 180)
  if(MATTER_SANITIZE)
    if(WIN32)
      set(leaks 0)
    else()
      set(leaks 1)
    endif()
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "ASAN_OPTIONS=detect_leaks=${leaks}:halt_on_error=1;UBSAN_OPTIONS=halt_on_error=1:print_stacktrace=1")
  endif()
endforeach()
