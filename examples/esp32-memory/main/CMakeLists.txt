get_filename_component(matter_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(port_tests)
if(CONFIG_MATTER_C_LAB_SELF_TESTS)
  foreach(test crypto_mbedtls spake2p credentials operational_cert attestation_cert im_read im_subscribe im_event_metadata)
    set(source "${matter_root}/tools/matter/tests/test_${test}.c")
    list(APPEND port_tests "${source}")
  endforeach()
endif()
idf_component_register(SRCS "app_main.c" ${port_tests} INCLUDE_DIRS "." PRIV_INCLUDE_DIRS "compat"
    REQUIRES matter_c esp_wifi esp_event esp_netif nvs_flash esp_timer bootloader_support)
if(CONFIG_MATTER_C_LAB_SELF_TESTS)
  foreach(test crypto_mbedtls spake2p credentials operational_cert attestation_cert im_read im_subscribe im_event_metadata)
    set_source_files_properties("${matter_root}/tools/matter/tests/test_${test}.c"
        PROPERTIES COMPILE_DEFINITIONS "main=matter_lab_test_${test}")
  endforeach()
endif()
target_compile_options(${COMPONENT_LIB} PRIVATE -fstack-usage)
if(NOT CONFIG_MATTER_C_PROFILE_ALLOCATIONS)
    message(FATAL_ERROR "The memory laboratory requires CONFIG_MATTER_C_PROFILE_ALLOCATIONS=y")
endif()
if(MATTER_C_LAB_CONFIG_DIR)
    target_include_directories(${COMPONENT_LIB} PRIVATE "${MATTER_C_LAB_CONFIG_DIR}")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE MATTER_C_LAB_WIFI_CONFIG=1)
endif()
